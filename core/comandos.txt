Write-Host "=== MIGRATE DOWNLOADS + CLEAN ===" -ForegroundColor Cyan

$base = "$env:SystemDrive\Users"

# Perfil atual
$currentPath = $env:USERPROFILE
$currentName = Split-Path $currentPath -Leaf
$currentDownloads = Join-Path $currentPath "Downloads"

Write-Host "Current profile: $currentName"

# Extrai número
if ($currentName -match "TEMP\.APS\.(\d+)") {
    $num = [int]$matches[1]

    # Perfil anterior
    $prevName = "TEMP.APS." + ($num - 1).ToString("D3")
    $prevPath = Join-Path $base $prevName
    $prevDownloads = Join-Path $prevPath "Downloads"

    Write-Host "Previous profile: $prevName"

    # =========================
    # COPIAR DOWNLOADS
    # =========================
    if (Test-Path $prevDownloads) {
        Write-Host "Copying Downloads from $prevName..."

        try {
            Copy-Item "$prevDownloads\*" -Destination $currentDownloads `
                -Recurse -Force -ErrorAction SilentlyContinue

            Write-Host "Copy done ✅" -ForegroundColor Green
        }
        catch {
            Write-Host "Some files could not be copied (locked)" -ForegroundColor Yellow
        }
    } else {
        Write-Host "No Downloads folder in previous profile"
    }

    # =========================
    # LIMPAR ANTERIOR
    # =========================
    if (Test-Path $prevPath) {
        Write-Host "Cleaning previous profile..."

        try {
            Get-ChildItem $prevPath -Force -ErrorAction SilentlyContinue |
                Remove-Item -Recurse -Force -ErrorAction SilentlyContinue

            Write-Host "Cleanup done ✅" -ForegroundColor Green
        }
        catch {
            Write-Host "Partial cleanup (files locked)" -ForegroundColor Yellow
        }
    }

} else {
    Write-Host "Not running in TEMP.APS profile" -ForegroundColor Red
}

Write-Host "=== DONE ==="
